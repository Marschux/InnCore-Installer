# =============================================================================
# AZEROTHCORE - AUTH SERVER COMMAND REFERENCE
# =============================================================================

authmenu          # Opens the main authserver management menu
authstart         # Starts the authserver
authstop          # Stops the authserver
authrestart       # Restarts the authserver
authstatus        # Shows the current authserver status

authmonitor       # Opens the live authserver console
authlogs          # Shows the last 300 authserver console lines
authlogsfull      # Shows full authserver tmux scrollback

authautostart     # Manages authserver autostart
authupdate        # Updates AzerothCore source for the authserver
authbuild         # Builds and installs the authserver
authcleanbuild    # Safely cleans the authserver build directory

auth-paths        # Opens auth_paths.conf
auth-conf         # Opens authserver.conf
auth-values       # Opens values_auth.conf
auth-patch        # Applies the authserver config patch

realmlist         # Opens the realmlist & MOTD manager

accounts          # Opens the account manager (add, edit, set GM level)

authincrement     # Manages the account table AUTO_INCREMENT value
authcommands      # Shows all configured authserver aliases
h                 # Opens htop
glance            # Opens Glances system monitoring
